let currentAdmissionNo = ""; async function requestOtp() { const admInput = document.getElementById('admissionInput'); const errBox = document.getElementById('authError'); currentAdmissionNo = admInput.value.trim().toUpperCase(); if (!currentAdmissionNo) { errBox.innerText = "Kripya Admission Number darj karein."; errBox.classList.remove('hidden'); return; } errBox.classList.add('hidden'); try { const res = await fetch('/send-otp', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ admission_no: currentAdmissionNo }) }); const data = await res.json(); if (data.success) { document.getElementById('stepAdmission').classList.add('hidden'); document.getElementById('stepOtp').classList.remove('hidden'); // Demo / Testing hint (waise email par jata hai) document.getElementById('otpHint').innerText = `Demo OTP: ${data.demo_otp}`; } else { errBox.innerText = data.message; errBox.classList.remove('hidden'); } } catch (e) { errBox.innerText = "Server error while generating OTP."; errBox.classList.remove('hidden'); } } async function submitOtp() { const otpInput = document.getElementById('otpInput'); const errBox = document.getElementById('authError'); const otp = otpInput.value.trim(); if (otp.length < 6) { errBox.innerText = "Kripya 6-digit OTP enter karein."; errBox.classList.remove('hidden'); return; } try { const res = await fetch('/verify-otp', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ admission_no: currentAdmissionNo, otp: otp }) }); const data = await res.json(); if (data.success) { isAuthenticated = true; document.getElementById('loginModal').classList.add('hidden'); document.getElementById('studentBadge').innerText = data.student_name; appendMsg(`✅ ${data.message}! Navigation unlock ho gaya hai.`, 'bot'); } else { errBox.innerText = data.message; errBox.classList.remove('hidden'); } } catch (e) { errBox.innerText = "Verification failed. Kripya dobara try karein."; errBox.classList.remove('hidden'); } } function backToAdmission() { document.getElementById('stepOtp').classList.add('hidden'); document.getElementById('stepAdmission').classList.remove('hidden'); document.getElementById('authError').classList.add('hidden'); }
